Acest fișier trebuie salvat în repository ca:  .github/workflows/pages.yml
(Windows nu lasă tools-urile externe să scrie direct în .github, de aceea e copiat aici ca .txt.
 Varianta gata pusă la locul ei se află în arhiva .zip a aplicației.)

name: Publică pe GitHub Pages

on:
  push:
    branches: [main]
  workflow_dispatch:

permissions:
  contents: read
  pages: write
  id-token: write

concurrency:
  group: pages
  cancel-in-progress: false

jobs:
  deploy:
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    runs-on: ubuntu-latest
    steps:
      - name: Ia codul
        uses: actions/checkout@v4

      - name: Configurează Pages
        uses: actions/configure-pages@v5

      - name: Încarcă site-ul (tot repo-ul, nu e nimic de compilat)
        uses: actions/upload-pages-artifact@v3
        with:
          path: '.'

      - name: Publică
        id: deployment
        uses: actions/deploy-pages@v4
